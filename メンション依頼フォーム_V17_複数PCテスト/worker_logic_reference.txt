$ErrorActionPreference="Stop"

$pendingPath=$env:PENDING_PATH
$csvFolderArg=$env:CSV_FOLDER_ARG
$resultPath=$env:RESULT_PATH
$retryMin=450
$retryMax=1250
$maxRetrySeconds=300

function Decode([string]$s){
    return [Uri]::UnescapeDataString($s)
}

function Delay{
    $min=[Math]::Max(50,$retryMin)
    $max=[Math]::Max($min+1,$retryMax+1)
    Start-Sleep -Milliseconds (Get-Random -Minimum $min -Maximum $max)
}

function Read-Pending([string]$Path){
    if(-not(Test-Path -LiteralPath $Path)){throw "PENDING_NOT_FOUND"}

    $text=Get-Content -LiteralPath $Path -Raw
    $rows=$text -replace "`r`n","`n" -replace "`r","`n" -split "`n"

    $format=""
    $baseName=""
    $requestId=""
    $created=""
    $pendingCsvFolder=""
    $lines=New-Object System.Collections.Generic.List[string]

    foreach($row in $rows){
        if($row.StartsWith("FORMAT=")){$format=$row.Substring(7)}
        elseif($row.StartsWith("BASE=")){$baseName=Decode $row.Substring(5)}
        elseif($row.StartsWith("REQUEST_ID=")){$requestId=Decode $row.Substring(11)}
        elseif($row.StartsWith("CREATED=")){$created=Decode $row.Substring(8)}
        elseif($row.StartsWith("CSV_FOLDER=")){$pendingCsvFolder=Decode $row.Substring(11)}
        elseif($row.StartsWith("LINE=")){[void]$lines.Add((Decode $row.Substring(5)))}
    }

    if($format -ne "MENTION_REQUEST_PENDING_V3"){throw "PENDING_FORMAT_INVALID"}
    if([string]::IsNullOrWhiteSpace($baseName)){throw "BASE_MISSING"}
    if([string]::IsNullOrWhiteSpace($requestId)){throw "REQUEST_ID_MISSING"}
    if($lines.Count -eq 0){throw "LINES_MISSING"}

    return [PSCustomObject]@{
        BaseName=$baseName
        RequestId=$requestId
        Created=$created
        CsvFolder=$pendingCsvFolder
        Lines=$lines
    }
}

function Safe-Base([string]$Text){
    foreach($c in [IO.Path]::GetInvalidFileNameChars()){
        $Text=$Text.Replace([string]$c,"_")
    }
    return $Text
}

function Date-Key([string]$Created){
    if($Created -match "^(\d{4})/(\d{2})/(\d{2})"){
        return $Matches[1]+$Matches[2]+$Matches[3]
    }
    return Get-Date -Format "yyyyMMdd"
}

function Csv-Path([string]$Folder,[string]$Base,[string]$Created){
    return Join-Path $Folder ((Safe-Base $Base)+"_"+(Date-Key $Created)+".csv")
}

function Acquire-Lock([string]$Path){
    $s=Get-Date
    while($true){
        try{
            return [IO.File]::Open($Path,[IO.FileMode]::OpenOrCreate,[IO.FileAccess]::ReadWrite,[IO.FileShare]::None)
        }catch{
            if(((Get-Date)-$s).TotalSeconds -ge 5){throw "LOCK_TIMEOUT"}
            Delay
        }
    }
}

function Parser-Path([string]$Path){
    Add-Type -AssemblyName Microsoft.VisualBasic -ErrorAction SilentlyContinue
    $p=New-Object Microsoft.VisualBasic.FileIO.TextFieldParser($Path,[Text.Encoding]::GetEncoding(932))
    $p.TextFieldType=[Microsoft.VisualBasic.FileIO.FieldType]::Delimited
    $p.SetDelimiters(",")
    $p.HasFieldsEnclosedInQuotes=$true
    return $p
}

function Parse-Record([string]$Record){
    Add-Type -AssemblyName Microsoft.VisualBasic -ErrorAction SilentlyContinue
    $reader=New-Object IO.StringReader($Record)
    $p=New-Object Microsoft.VisualBasic.FileIO.TextFieldParser($reader)
    $p.TextFieldType=[Microsoft.VisualBasic.FileIO.FieldType]::Delimited
    $p.SetDelimiters(",")
    $p.HasFieldsEnclosedInQuotes=$true
    try{return $p.ReadFields()}
    finally{$p.Close();$reader.Dispose()}
}

function Existing([string]$Path,[string]$RequestId){
    $h=@{}
    if(-not(Test-Path -LiteralPath $Path)){return $h}

    $p=Parser-Path $Path
    try{
        $isHeader=$true
        while(-not $p.EndOfData){
            $f=$p.ReadFields()
            if($isHeader){
                $isHeader=$false
                continue
            }
            if($f -and $f.Length -ge 5 -and [string]$f[1] -eq $RequestId){
                $no=[string]$f[4]
                if(-not[string]::IsNullOrWhiteSpace($no)){$h[$no]=$true}
            }
        }
    }finally{$p.Close()}
    return $h
}

function Pending-Map($Lines){
    $map=@{}
    foreach($line in $Lines){
        $f=Parse-Record $line
        if(-not $f -or $f.Length -lt 5){throw "PENDING_CSV_PARSE_FAILED"}
        $no=[string]$f[4]
        if([string]::IsNullOrWhiteSpace($no)){throw "REQUEST_NO_MISSING"}
        $map[$no]=$line
    }
    return $map
}

function Write-WithLock([string]$Folder,[string]$Base,[string]$RequestId,[string]$Created,$Lines){
    if(-not(Test-Path -LiteralPath $Folder)){
        New-Item -ItemType Directory -Path $Folder -Force|Out-Null
    }

    $csv=Csv-Path $Folder $Base $Created
    $lock=$csv+".lock"
    $ls=$null

    try{
        $ls=Acquire-Lock $lock
        $exists=Test-Path -LiteralPath $csv
        $existing=Existing $csv $RequestId
        $map=Pending-Map $Lines
        $missing=New-Object System.Collections.Generic.List[string]

        foreach($no in $map.Keys){
            if(-not $existing.ContainsKey($no)){[void]$missing.Add([string]$map[$no])}
        }

        if($missing.Count -eq 0){return "DUPLICATE_SKIP"}

        $partial=($existing.Count -gt 0)
        $enc=[Text.Encoding]::GetEncoding(932)
        $w=New-Object IO.StreamWriter($csv,$true,$enc)

        try{
            if(-not $exists){
                $header=[Uri]::UnescapeDataString("%E9%80%81%E4%BF%A1%E6%97%A5%E6%99%82%2CRequestID%2C%E6%8B%A0%E7%82%B9%2C%E4%BE%9D%E9%A0%BC%E8%80%85%2C%E4%BE%9D%E9%A0%BC%E7%95%AA%E5%8F%B7%2C%E7%B5%84%E7%B9%94%2CCA%E5%90%8D%2C%E4%BB%A3%E7%90%86CA%E7%B5%84%E7%B9%94%2C%E4%BB%A3%E7%90%86CA1%2C%E4%BB%A3%E7%90%86CA2%2C%E4%BB%A3%E7%90%86CA3%2C%E3%83%A1%E3%83%BC%E3%83%AB%E3%83%A1%E3%83%A2%2C%E5%87%A6%E7%90%86%E6%97%A5%E6%99%82%2C%E6%9C%9F%E6%97%A5%2C%E6%99%82%E7%9F%AD%2C%E8%87%B3%E6%80%A5%2C%E3%82%BF%E3%82%A4%E3%83%97")
                $w.WriteLine($header)
            }
            foreach($line in $missing){$w.WriteLine($line)}
            $w.Flush()
            try{$w.BaseStream.Flush($true)}catch{}
        }finally{$w.Dispose()}

        $verified=Existing $csv $RequestId
        foreach($no in $map.Keys){
            if(-not $verified.ContainsKey($no)){throw "VERIFY_FAILED"}
        }

        if($partial){return "RECOVERED_PARTIAL"}
        return "SUCCESS"
    }finally{
        if($ls){$ls.Dispose()}
        Remove-Item -LiteralPath $lock -Force -ErrorAction SilentlyContinue
    }
}

function Remove-Pending([string]$Path){
    if(-not(Test-Path -LiteralPath $Path)){return $true}
    for($i=0;$i -lt 5;$i++){
        try{
            Remove-Item -LiteralPath $Path -Force -ErrorAction Stop
            return $true
        }catch{Delay}
    }
    return $false
}

$started=Get-Date
$retry=0
$lockWait=0
$status=""
$lastError=""
$pendingRemoved=$false
$pending=$null

try{
    $pending=Read-Pending $pendingPath

    $effective=[string]$csvFolderArg
    if([string]::IsNullOrWhiteSpace($effective)){$effective=[string]$pending.CsvFolder}
    if([string]::IsNullOrWhiteSpace($effective)){throw "CSV_FOLDER_MISSING"}

    while($true){
        try{
            $status=Write-WithLock $effective $pending.BaseName $pending.RequestId $pending.Created $pending.Lines
            break
        }catch{
            $retry++
            $lastError=$_.Exception.Message

            if(((Get-Date)-$started).TotalSeconds -ge $maxRetrySeconds){
                $status="RETRY_EXHAUSTED"
                break
            }
            Delay
        }
    }

    if($status -ne "RETRY_EXHAUSTED"){
        $pendingRemoved=Remove-Pending $pendingPath
        if(-not $pendingRemoved){$status="PENDING_DELETE_FAILED"}
    }
}catch{
    $status="FATAL"
    $lastError=$_.Exception.Message
}

$dur=[Math]::Round(((Get-Date)-$started).TotalMilliseconds,2)
$lineCount=0
if($pending){$lineCount=$pending.Lines.Count}
$out=@($status,$retry,$dur,$pendingRemoved,$lineCount,$lastError)-join "|"
[IO.File]::WriteAllText($resultPath,$out,[Text.Encoding]::UTF8)
